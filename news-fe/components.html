Data Grid

const grid = UI.grid({
    data: articles,                  // array of objects (each needs unique `id`)
    columns: [
        { key: 'id',     label: '#',       width: 60 },
        { key: 'title',  label: 'Title',   sortable: true },
        { key: 'score',  label: 'Score',   align: 'right',
          render: (v) => `<strong class="text-primary">${v}%</strong>` },
        { key: 'actions', label: '',        sortable: false,
          render: (_, row) => UI.button({ label: 'Edit', variant: 'soft', size: 'sm' }) },
    ],
    pageSize: 5,
    pageSizeOptions: [5, 10, 25],
    searchable: true,
    selectable: true,
    onRowClick: (row) => UI.alert.info({ title: row.title }),
    onSelectionChange: (rows) => console.log(rows.length),
    toolbar: '<button>Add</button>', // custom right-side content
});

// Methods
grid.setData(newArray);
grid.getSelected();       // Array of selected rows
grid.setLoading(true);
grid.setSort('score', 'desc');
grid.setSearch('legal');
grid.goToPage(2);
grid.refresh();
